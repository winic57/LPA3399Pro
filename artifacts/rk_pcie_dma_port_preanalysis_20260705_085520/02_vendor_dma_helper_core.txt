30:/* dma transfer */
31:/*
32: * Write buffer format
33: * 0	     4               8	       0xc	0x10	SZ_1M
34: * ------------------------------------------------------
35: * |0x12345678|local idx(0-7)|data size|reserve	|data	|
36: * ------------------------------------------------------
37: *
38: * Byte 3-0: Receiver check if a valid data package arrived
39: * Byte 7-4: As a index for data rcv ack buffer
40: * Byte 11-8: Actual data size
41: *
42: * Data rcv ack buffer format
43: * 0		4B
44: * --------------
45: * |0xdeadbeef	|
46: * --------------
47: *
48: * Data free ack buffer format
49: * 0		4B
50: * --------------
51: * |0xcafebabe	|
52: * --------------
53: *
54: *	RC		EP
55: * -	---------	---------
56: * |	|  1MB	|	|	|
57: * |	|------	|	|	|
58: * |	|	|	|	|
59: * |	|	|	|	|
60: * 8MB	|wr buf	|  ->	|rd buf	|
61: * |	|	|	|	|
62: * |	|	|	|	|
63: * |	|	|	|	|
64: * -	---------	---------
65: * |	|  1MB	|	|	|
66: * |	|------	|	|	|
67: * |	|	|	|	|
68: * |	|	|	|	|
69: * 8MB	|rd buf	|  <-	|wr buf	|
70: * |	|	|	|	|
71: * |	|	|	|	|
72: * |	|	|	|	|
73: * -	---------	---------
74: * |	|  4B	|	|	|
75: * |	|------	|	|	|
76: * |	|	|	|	|
77: * 32B	|	|	|	|
78: * |	|scan	|  <-	|data	|
79: * |	|	|	|rcv	|
80: * |	|	|	|ack	|
81: * |	|	|	|send	|
82: * -	---------	---------
83: * |	|  4B	|	|	|
84: * |	|------	|	|	|
85: * |	|	|	|	|
86: * 32B	|data	|  ->	|scan	|
87: * |	|rcv	|	|	|
88: * |	|ack	|	|	|
89: * |	|send	|	|	|
90: * |	|	|	|	|
91: * -	---------	---------
92: * |	|  4B	|	|	|
93: * |	|------	|	|	|
94: * |	|	|	|	|
95: * 32B	|	|	|	|
96: * |	|scan	|  <-	|data	|
97: * |	|	|	|free	|
98: * |	|	|	|ack	|
99: * |	|	|	|send	|
100: * -	---------	---------
101: * |	|4B	|	|	|
102: * |	|------	|	|	|
103: * |	|	|	|	|
104: * 32B	|data	|  ->	|scan	|
105: * |	|free	|	|	|
106: * |	|ack	|	|	|

108: * |	|	|	|	|

142:static void rk_pcie_prepare_dma(struct dma_trx_obj *obj,
143:			unsigned int idx, unsigned int bus_idx,
144:			unsigned int local_idx, size_t buf_size,
145:			enum transfer_type type)
146:{
147:	struct device *dev = obj->dev;
148:	phys_addr_t local, bus;
149:	void *virt;
150:	u32 *desc;
151:	unsigned long flags;
152:	struct dma_table *table = NULL;
153:
154:	switch (type) {
155:	case PCIE_DMA_DATA_SND:
156:		table = obj->table[PCIE_DMA_DATA_SND_TABLE_OFFSET + local_idx];
157:		table->type = PCIE_DMA_DATA_SND;
158:		local = obj->mem_start + local_idx * PCIE_DMA_BUF_SIZE;
159:		virt = obj->mem_base + local_idx * PCIE_DMA_BUF_SIZE;
160:		bus = obj->mem_start + bus_idx * PCIE_DMA_BUF_SIZE;
161:
162:		if (!is_rc(obj)) {
163:			local += PCIE_DMA_RD_BUF_SIZE;
164:			virt += PCIE_DMA_RD_BUF_SIZE;
165:			bus += PCIE_DMA_WR_BUF_SIZE;
166:		}
167:
168:		dma_sync_single_for_device(dev, local, buf_size, DMA_TO_DEVICE);
169:
170:		writel(PCIE_DMA_DATA_CHECK, virt + PCIE_DMA_SET_DATA_CHECK_POS);
171:		writel(local_idx, virt + PCIE_DMA_SET_LOCAL_IDX_POS);
172:		writel(buf_size, virt + PCIE_DMA_SET_BUF_SIZE_POS);
173:
174:		buf_size = SZ_1M;
175:		break;
176:	case PCIE_DMA_DATA_RCV_ACK:
177:		table = obj->table[PCIE_DMA_DATA_RCV_ACK_TABLE_OFFSET + idx];
178:		table->type = PCIE_DMA_DATA_RCV_ACK;
179:		local = obj->mem_start + PCIE_DMA_ACK_BASE + idx * NODE_SIZE;
180:		virt = obj->mem_base + PCIE_DMA_ACK_BASE + idx * NODE_SIZE;
181:
182:		if (is_rc(obj)) {
183:			local += PCIE_DMA_ACK_BLOCK_SIZE;
184:			virt += PCIE_DMA_ACK_BLOCK_SIZE;
185:		}
186:		bus = local;
187:		writel(PCIE_DMA_DATA_ACK_CHECK, virt);
188:		break;
189:	case PCIE_DMA_DATA_FREE_ACK:
190:		table = obj->table[PCIE_DMA_DATA_FREE_ACK_TABLE_OFFSET + idx];
191:		table->type = PCIE_DMA_DATA_FREE_ACK;
192:		local = obj->mem_start + PCIE_DMA_ACK_BASE + idx * NODE_SIZE;
193:		virt = obj->mem_base + PCIE_DMA_ACK_BASE + idx * NODE_SIZE;
194:
195:		if (is_rc(obj)) {
196:			local += 3 * PCIE_DMA_ACK_BLOCK_SIZE;
197:			virt += 3 * PCIE_DMA_ACK_BLOCK_SIZE;
198:		} else {
199:			local += 2 * PCIE_DMA_ACK_BLOCK_SIZE;
200:			virt += 2 * PCIE_DMA_ACK_BLOCK_SIZE;
201:		}
202:		bus = local;
203:		writel(PCIE_DMA_DATA_FREE_ACK_CHECK, virt);
204:		break;
205:	default:
206:		dev_err(dev, "type = %d not support\n", type);
207:		return;
208:	}
209:
210:	if (is_rc(obj)) {
211:		desc = table->descs;
212:		*(desc + 0) = (u32)(local & 0xffffffff);
213:		*(desc + 1) = (u32)(local >> 32);
214:		*(desc + 2) = (u32)(bus & 0xffffffff);
215:		*(desc + 3) = (u32)(bus >> 32);
216:		*(desc + 4) = 0;
217:		*(desc + 5) = 0;
218:		*(desc + 6) = buf_size;
219:		*(desc + 7) = 0;
220:		*(desc + 8) = 0;

413:static long rk_pcie_misc_ioctl(struct file *filp, unsigned int cmd,
414:					unsigned long arg)
415:{
416:	struct dma_trx_obj *obj = filp->private_data;
417:	struct device *dev = obj->dev;
418:	union pcie_dma_ioctl_param msg;
419:	union pcie_dma_ioctl_param msg_to_user;
420:	phys_addr_t addr;
421:	void __user *uarg = (void __user *)arg;
422:	int ret;
423:
424:	if (copy_from_user(&msg, uarg, sizeof(msg)) != 0) {
425:		dev_err(dev, "failed to copy argument into kernel space\n");
426:		return -EFAULT;
427:	}
428:
429:	switch (cmd) {
430:	case PCIE_DMA_START:
431:		test_and_clear_bit(msg.in.l_widx, &obj->local_write_available);
432:		test_and_clear_bit(msg.in.r_widx, &obj->remote_write_available);
433:		obj->loop_count++;
434:		break;
435:	case PCIE_DMA_GET_LOCAL_READ_BUFFER_INDEX:
436:		msg_to_user.lra = obj->local_read_available;
437:		addr = obj->mem_start;
438:		if (is_rc(obj))
439:			addr += PCIE_DMA_WR_BUF_SIZE;
440:		/* by kernel auto or by user to invalidate cache */
441:		dma_sync_single_for_cpu(dev, addr, PCIE_DMA_RD_BUF_SIZE,
442:					DMA_FROM_DEVICE);
443:		ret = copy_to_user(uarg, &msg_to_user, sizeof(msg));
444:		if (ret) {
445:			dev_err(dev, "failed to get read buffer index\n");
446:			return -EFAULT;
447:		}
448:		break;
449:	case PCIE_DMA_SET_LOCAL_READ_BUFFER_INDEX:
450:		test_and_clear_bit(msg.in.idx, &obj->local_read_available);
451:		break;
452:	case PCIE_DMA_GET_LOCAL_REMOTE_WRITE_BUFFER_INDEX:
453:		msg_to_user.out.lwa = obj->local_write_available;
454:		msg_to_user.out.rwa = obj->remote_write_available;
455:		ret = copy_to_user(uarg, &msg_to_user, sizeof(msg));
456:		if (ret) {
457:			dev_err(dev, "failed to get write buffer index\n");
458:			return -EFAULT;
459:		}
460:		break;
461:	case PCIE_DMA_SYNC_BUFFER_FOR_CPU:
462:		addr = obj->mem_start + msg.in.idx * PCIE_DMA_BUF_SIZE;
463:		if (is_rc(obj))
464:			addr += PCIE_DMA_WR_BUF_SIZE;
465:		dma_sync_single_for_cpu(dev, addr, PCIE_DMA_BUF_SIZE,
466:					DMA_FROM_DEVICE);
467:		break;
468:	case PCIE_DMA_WAIT_TRANSFER_COMPLETE:
469:		reinit_completion(&obj->done);
470:		ret = wait_for_completion_interruptible(&obj->done);
471:		if (WARN_ON(ret)) {
472:			pr_info("failed to wait complete\n");
473:			return ret;
474:		}
475:
476:		obj->loop_count = 0;
477:		break;
478:	case PCIE_DMA_SET_LOOP_COUNT:
479:		obj->loop_count_threshold = msg.count;
480:		pr_info("threshold = %d\n", obj->loop_count_threshold);
481:		break;
482:	default:
483:		pr_info("%s, %d, cmd : %x not support\n", __func__, __LINE__,
484:			cmd);
485:		return -EFAULT;
486:	}
487:
488:	if (cmd == PCIE_DMA_START ||
489:		cmd == PCIE_DMA_SET_LOCAL_READ_BUFFER_INDEX) {
490:		rk_pcie_prepare_dma(obj, msg.in.idx, msg.in.r_widx,
491:				    msg.in.l_widx, msg.in.size, msg.in.type);
492:		queue_work(obj->dma_trx_wq, &obj->dma_trx_work);
493:	}
494:
495:	return 0;
496:}
